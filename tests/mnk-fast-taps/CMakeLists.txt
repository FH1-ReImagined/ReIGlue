cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(mnk_fast_taps LANGUAGES CXX)
  find_package(rexglue 0.10.0 REQUIRED CONFIG)
  enable_testing()
  set(mnk_runtime rex::runtime)
else()
  set(mnk_runtime rexruntime)
endif()
add_executable(mnk_fast_taps_test mnk_fast_taps_test.cpp)
target_compile_features(mnk_fast_taps_test PRIVATE cxx_std_23)
target_link_libraries(mnk_fast_taps_test PRIVATE ${mnk_runtime})
if(MSVC)
  target_compile_options(mnk_fast_taps_test PRIVATE /UNDEBUG /EHsc)
else()
  target_compile_options(mnk_fast_taps_test PRIVATE -UNDEBUG)
endif()
add_test(NAME mnk_fast_taps COMMAND mnk_fast_taps_test)
set_tests_properties(mnk_fast_taps PROPERTIES TIMEOUT 10)
if(WIN32)
  add_custom_command(TARGET mnk_fast_taps_test POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:${mnk_runtime}>"
      "$<TARGET_FILE_DIR:mnk_fast_taps_test>")
endif()
